cmake_minimum_required(VERSION 3.22)
project(glassmic C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-Wall -Wextra -Wno-unused-parameter -ffunction-sections -fdata-sections)
add_link_options(-Wl,--gc-sections -Wl,-z,max-page-size=16384)
set(THIRD ${CMAKE_CURRENT_SOURCE_DIR}/../third_party)
set(SH ${THIRD}/shadowhook-v2.0.1/shadowhook/src/main/cpp)
file(GLOB SH_SOURCES ${SH}/*.c ${SH}/arch/arm64/*.c ${SH}/arch/arm64/*.S ${SH}/common/*.c ${SH}/third_party/*/*.c)
list(FILTER SH_SOURCES EXCLUDE REGEX "sh_jni.c$")
add_library(shadowhook_embedded STATIC ${SH_SOURCES})
target_include_directories(shadowhook_embedded PUBLIC ${SH}/include PRIVATE ${SH} ${SH}/arch/arm64 ${SH}/common ${SH}/third_party/xdl ${SH}/third_party/bsd ${SH}/third_party/lss)
target_compile_options(shadowhook_embedded PRIVATE -fvisibility=hidden -w)
target_link_libraries(shadowhook_embedded PUBLIC log dl)
add_library(glassmic_decoder STATIC decoder.cpp)
target_include_directories(glassmic_decoder PRIVATE ${THIRD})
target_compile_options(glassmic_decoder PRIVATE -w)
add_executable(glassmicd daemon.cpp)
target_link_libraries(glassmicd glassmic_decoder log)
add_executable(glassmicctl ctl.cpp)
add_library(glassmic SHARED zygisk_module.cpp)
target_include_directories(glassmic PRIVATE ${THIRD})
target_compile_options(glassmic PRIVATE -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_options(glassmic PRIVATE -Wl,--exclude-libs,ALL)
target_link_libraries(glassmic android dl)
add_library(glassmic-backend SHARED capture.cpp opensl.cpp)
target_include_directories(glassmic-backend PRIVATE ${THIRD})
target_compile_options(glassmic-backend PRIVATE -fvisibility=hidden -fvisibility-inlines-hidden)
# ShadowHook dlopen()s libshadowhook_nothing.so while hooking the linker; the
# wrapper in capture.cpp loads the module's copy by descriptor.
target_link_options(glassmic-backend PRIVATE -Wl,--exclude-libs,ALL -Wl,--wrap=dlopen)
target_link_libraries(glassmic-backend shadowhook_embedded aaudio OpenSLES log)
add_library(shadowhook_nothing SHARED ${SH}/nothing/sh_nothing.c)
target_compile_options(shadowhook_nothing PRIVATE -w)
target_link_options(shadowhook_nothing PRIVATE -Wl,--strip-all)
add_executable(glassmic-selftest selftest.cpp)
target_link_libraries(glassmic-selftest glassmic_decoder dl)
